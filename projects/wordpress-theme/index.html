Crumb — custom WordPress theme

A classic WordPress theme written from scratch in PHP: template hierarchy, menus, custom logo, widgets and escaped output. Downloadable as a zip.

The problem

Most small businesses run on WordPress, and many job ads ask for someone who can build or edit a theme, not just install one. A heavy page-builder theme is slow and hard to maintain for a simple bakery site.

The solution

I wrote a lightweight classic theme by hand, using WordPress's own functions the way the official Theme Handbook recommends. The owner edits pages, posts, the menu and the logo from the normal WordPress dashboard; no page builder, no jQuery.

Key features

  • Standard template files: style.css header, functions.php, header.php, footer.php, index.php, page.php, single.php
  • Theme supports: title tag, featured images, custom logo, HTML5 markup, responsive embeds, wide alignment
  • Two menu locations (primary and footer) plus a footer widget area for opening hours and address
  • Accessibility: skip link, aria-label on navigation, keyboard-friendly mobile menu with aria-expanded, visible focus styles
  • Security: every output escaped with esc_html(), esc_attr(), esc_url() or wp_kses_post(); no direct file access
  • Translation-ready (crumb text domain) and a tiny inline script loaded in the footer

Quality checks

  • Responsive from 320px phones to wide desktop screens
  • Keyboard navigation and visible focus states
  • Labelled form fields and readable colour contrast
  • No frameworks: plain HTML, CSS and JavaScript, plus PHP

functions.php
<?php
/**
 * Crumb theme functions.
 *
 * @package Crumb
 */

if ( ! defined( 'ABSPATH' ) ) {
	exit; // No direct access.
}

define( 'CRUMB_VERSION', '1.0.0' );

/**
 * Theme setup: features, menus, translations.
 */
function crumb_setup() {
	load_theme_textdomain( 'crumb', get_template_directory() . '/languages' );

	add_theme_support( 'title-tag' );
	add_theme_support( 'post-thumbnails' );
	add_theme_support( 'automatic-feed-links' );
	add_theme_support( 'responsive-embeds' );
	add_theme_support( 'align-wide' );
	add_theme_support( 'wp-block-styles' );
	add_theme_support( 'html5', array( 'search-form', 'comment-form', 'comment-list', 'gallery', 'caption', 'style', 'script', 'navigation-widgets' ) );
	add_theme_support(
		'custom-logo',
		array(
			'height'      => 80,
			'width'       => 240,
			'flex-height' => true,
			'flex-width'  => true,
		)
	);

	register_nav_menus(
		array(
			'primary' => __( 'Primary menu', 'crumb' ),
			'footer'  => __( 'Footer menu', 'crumb' ),
		)
	);

	add_image_size( 'crumb-card', 720, 450, true );
}
add_action( 'after_setup_theme', 'crumb_setup' );

/**
 * Content width for embeds and images.
 */
function crumb_content_width() {
	$GLOBALS['content_width'] = 760;
}
add_action( 'after_setup_theme', 'crumb_content_width', 0 );

/**
 * Footer widget area.
 */
function crumb_widgets_init() {
	register_sidebar(
		array(
			'name'          => __( 'Footer', 'crumb' ),
			'id'            => 'footer-1',
			'description'   => __( 'Widgets shown in the footer (opening hours, address, etc.).', 'crumb' ),
			'before_widget' => '<section id="%1$s" class="widget %2$s">',
			'after_widget'  => '</section>',
			'before_title'  => '<h2 class="widget-title">',
			'after_title'   => '</h2>',
		)
	);
}
add_action( 'widgets_init', 'crumb_widgets_init' );

/**
 * Styles and scripts. The menu script is tiny and loaded in the footer.
 */
function crumb_assets() {
	wp_enqueue_style( 'crumb-style', get_stylesheet_uri(), array(), CRUMB_VERSION );

	wp_register_script( 'crumb-navigation', '', array(), CRUMB_VERSION, true );
	wp_enqueue_script( 'crumb-navigation' );
	wp_add_inline_script(
		'crumb-navigation',
		"(function(){var n=document.getElementById('site-navigation');if(!n)return;var b=n.querySelector('.menu-toggle');if(!b)return;b.addEventListener('click',function(){var o=n.classList.toggle('toggled');b.setAttribute('aria-expanded',o?'true':'false');});})();"
	);

	if ( is_singular() && comments_open() && get_option( 'thread_comments' ) ) {
		wp_enqueue_script( 'comment-reply' );
	}
}
add_action( 'wp_enqueue_scripts', 'crumb_assets' );

/**
 * Shorter, cleaner excerpts on archive pages.
 *
 * @param int $length Default length.
 * @return int
 */
function crumb_excerpt_length( $length ) {
	return is_admin() ? $length : 28;
}
add_filter( 'excerpt_length', 'crumb_excerpt_length' );

/**
 * Prints the post date and author, escaped.
 */
function crumb_posted_on() {
	printf(
		/* translators: 1: post date, 2: author name. */
		esc_html__( 'Posted on %1$s by %2$s', 'crumb' ),
		'<time datetime="' . esc_attr( get_the_date( DATE_W3C ) ) . '">' . esc_html( get_the_date() ) . '</time>',
		esc_html( get_the_author() )
	);
}
header.php
<?php
/**
 * Site header.
 *
 * @package Crumb
 */
?>
<!doctype html>
<html <?php language_attributes(); ?>>
<head>
	<meta charset="<?php bloginfo( 'charset' ); ?>">
	<meta name="viewport" content="width=device-width, initial-scale=1">
	<?php wp_head(); ?>
</head>
<body <?php body_class(); ?>>
<?php wp_body_open(); ?>
<a class="skip-link" href="#primary"><?php esc_html_e( 'Skip to content', 'crumb' ); ?></a>

<header class="site-header">
	<div class="container">
		<div class="site-branding">
			<?php if ( has_custom_logo() ) : ?>
				<?php the_custom_logo(); ?>
			<?php else : ?>
				<p class="site-title"><a href="<?php echo esc_url( home_url( '/' ) ); ?>" rel="home"><?php bloginfo( 'name' ); ?></a></p>
			<?php endif; ?>
			<?php $crumb_description = get_bloginfo( 'description', 'display' ); ?>
			<?php if ( $crumb_description ) : ?>
				<p class="site-description"><?php echo esc_html( $crumb_description ); ?></p>
			<?php endif; ?>
		</div>

		<nav id="site-navigation" class="main-navigation" aria-label="<?php esc_attr_e( 'Primary', 'crumb' ); ?>">
			<button class="menu-toggle" aria-controls="primary-menu" aria-expanded="false"><?php esc_html_e( 'Menu', 'crumb' ); ?></button>
			<?php
			wp_nav_menu(
				array(
					'theme_location' => 'primary',
					'menu_id'        => 'primary-menu',
					'container'      => false,
					'fallback_cb'    => 'wp_page_menu',
					'depth'          => 2,
				)
			);
			?>
		</nav>
	</div>
</header>
single.php
<?php
/**
 * Template for a single blog post.
 *
 * @package Crumb
 */

get_header();
?>
<main id="primary" class="site-main">
	<div class="container">
		<?php
		while ( have_posts() ) :
			the_post();
			?>
			<article id="post-<?php the_ID(); ?>" <?php post_class( 'entry' ); ?>>
				<header>
					<h1 class="entry-title"><?php the_title(); ?></h1>
					<p class="entry-meta"><?php crumb_posted_on(); ?></p>
				</header>
				<?php if ( has_post_thumbnail() ) : ?>
					<div class="post-thumbnail"><?php the_post_thumbnail( 'large' ); ?></div>
				<?php endif; ?>
				<div class="entry-content">
					<?php the_content(); ?>
				</div>
				<footer class="entry-meta">
					<?php
					$crumb_categories = get_the_category_list( esc_html__( ', ', 'crumb' ) );
					if ( $crumb_categories ) {
						/* translators: %s: list of categories. */
						printf( '<p>' . esc_html__( 'Filed under: %s', 'crumb' ) . '</p>', wp_kses_post( $crumb_categories ) );
					}
					?>
				</footer>
			</article>

			<?php
			the_post_navigation(
				array(
					'prev_text' => '&larr; %title',
					'next_text' => '%title &rarr;',
				)
			);

			if ( comments_open() || get_comments_number() ) :
				comments_template();
			endif;
		endwhile;
		?>
	</div>
</main>
<?php
get_footer();
